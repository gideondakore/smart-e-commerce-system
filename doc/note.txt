# Start PostgreSQL
docker run -d \
  --name smart-ecommerce \
  -e POSTGRES_USER=spycon \
  -e POSTGRES_PASSWORD=postgressPassword12345 \
  -e POSTGRES_DB=ecommerce_db \
  -p 5432:5432 \
  postgres:latest

# Start pgAdmin
docker run -d \
  --name pgadmin \
  -e PGADMIN_DEFAULT_EMAIL=admin@admin.com \
  -e PGADMIN_DEFAULT_PASSWORD=admin \
  -p 5050:80 \
  dpage/pgadmin4

# Create a network manually
docker network create ecommerce-network
docker network connect ecommerce-network smart-ecommerce
docker network connect ecommerce-network pgadmin
